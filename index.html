Developer quickstart

Testnet config for viem, as shown in Dwellir’s guide. Mainnet chain ID was not published in the sources used here.

import { defineChain } from 'viem'

export const elysiumTestnet = defineChain({
  id: 99801,
  name: 'Elysium Testnet',
  nativeCurrency: { name: 'HYPE', symbol: 'HYPE', decimals: 18 },
  rpcUrls: { default: { http: ['https://testnet-rpc.elysium.kinetiq.xyz'] } },
  blockExplorers: { default: { name: 'Elysium Explorer',
    url: 'https://elysium.kinetiq.xyz/testnet-explorer' } },
})
